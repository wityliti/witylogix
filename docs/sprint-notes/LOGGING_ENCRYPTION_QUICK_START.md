## Quick Start: Logging

### Import
```typescript
import {
  createLogger,
  createRequestLoggerPlugin,
  LogLevel
} from '@witylogix/core/logging';
```

### Basic Logger Setup
```typescript
const logger = createLogger({
  level: LogLevel.INFO,
  service: 'my-service',
  pretty: process.env.NODE_ENV === 'development'
});

logger.info('Application started');
logger.warn('This is a warning', { userId: 123 });
logger.error('Something failed', new Error('Database error'));
```

### With Fastify
```typescript
import Fastify from 'fastify';
import { createLogger, createRequestLoggerPlugin } from '@witylogix/core/logging';

const fastify = Fastify();
const logger = createLogger({
  level: LogLevel.INFO,
  service: 'api-server'
});

// Register the plugin
await fastify.register(createRequestLoggerPlugin(logger));

fastify.get('/api/users/:id', (request, reply) => {
  const contextLogger = logger.withContext(getRequestContext(request));
  contextLogger.info('Fetching user');
});
```

### Request Context Binding
```typescript
import { getRequestContext } from '@witylogix/core/logging';

fastify.get('/orders/:id', async (request, reply) => {
  // Get request context (auto-generated by plugin)
  const context = getRequestContext(request);

  // Create logger with context
  const log = logger.withContext(context);

  // All logs will include traceId, duration, etc.
  log.info('Processing order', { orderId: request.params.id });
});
```

### Child Loggers
```typescript
// Create a child logger with bound values
const dbLogger = logger.child({
  component: 'database',
  poolSize: 20
});

dbLogger.debug('Query executed', { ms: 45 });
// Output includes: component, poolSize, and query metadata
```

### Sensitive Field Redaction (Automatic)
```typescript
logger.info('User login', {
  email: 'user@example.com',
  password: 'secret123',        // Automatically redacted
  apiKey: 'sk_prod_xyz',        // Automatically redacted
  creditCard: '4111-1111-1111'  // Automatically redacted
});

// Output:
// {
//   "message": "User login",
//   "email": "user@example.com",
//   "password": "[REDACTED]",
//   "apiKey": "[REDACTED]",
//   "creditCard": "[REDACTED]"
// }
```

## Quick Start: Encryption

### Import
```typescript
import {
  createCryptoService,
  createFieldEncryptor,
  EncryptionAlgorithm
} from '@witylogix/core/encryption';
```

### Setup (with environment variable)
```typescript
// .env
ENCRYPTION_MASTER_KEY="your-secure-master-key-at-least-32-characters"

// Code
const crypto = createCryptoService();
```

### Basic Encrypt/Decrypt
```typescript
const crypto = createCryptoService();

// Encrypt
const plaintext = "super-secret-api-key";
const encrypted = crypto.encrypt(plaintext);
console.log(encrypted);
// {
//   algorithm: 'aes-256-gcm',
//   iv: 'base64-encoded-iv',
//   authTag: 'base64-encoded-auth-tag',
//   data: 'base64-encoded-ciphertext',
//   keyId: 'default'
// }

// Decrypt
const decrypted = crypto.decrypt(encrypted);
console.log(decrypted); // "super-secret-api-key"
```

### Field-Level Encryption
```typescript
const crypto = createCryptoService();
const fieldEncryptor = createFieldEncryptor(crypto);

// Encrypt specific fields
const user = {
  id: 1,
  email: 'user@example.com',
  password: 'user-password',
  apiKey: 'sk_prod_xyz'
};

fieldEncryptor.encryptFields(user, ['password', 'apiKey']);
// user.password and user.apiKey are now encrypted

// Decrypt specific fields
fieldEncryptor.decryptFields(user, ['password', 'apiKey']);
// user.password and user.apiKey are decrypted
```

### Auto-Encrypt Sensitive Fields
```typescript
const fieldEncryptor = createFieldEncryptor(crypto);

const userData = {
  email: 'user@example.com',
  password: 'secret123',        // Will be auto-encrypted
  accessToken: 'token_xyz',     // Will be auto-encrypted
  name: 'John Doe'              // Won't be encrypted
};

// Automatically encrypts all sensitive fields
fieldEncryptor.encryptSensitiveFields(userData);

// Decrypt all sensitive fields
fieldEncryptor.decryptSensitiveFields(userData);
```

### Prisma Integration
```typescript
import { PrismaClient } from '@prisma/client';
import { createCryptoService, createFieldEncryptor } from '@witylogix/core/encryption';

const prisma = new PrismaClient();
const crypto = createCryptoService();
const fieldEncryptor = createFieldEncryptor(crypto);

// Add middleware
prisma.$use(fieldEncryptor.prismaMiddleware());

// Write: auto-encrypted
await prisma.user.create({
  data: {
    email: 'user@example.com',
    password: 'secret123'  // Auto-encrypted by middleware
  }
});

// Read: auto-decrypted
const user = await prisma.user.findUnique({ where: { id: 1 } });
console.log(user.password); // Decrypted plaintext
```

### Nested Field Encryption
```typescript
const fieldEncryptor = createFieldEncryptor(crypto);

const profile = {
  name: 'John Doe',
  settings: {
    webhookSecret: 'wh_secret_xyz',
    apiKey: 'sk_prod_abc'
  }
};

// Encrypt nested fields
fieldEncryptor.encryptFields(profile, [
  'settings.webhookSecret',
  'settings.apiKey'
]);

// Both nested fields are now encrypted
```

## Error Handling

### Logging Errors
```typescript
try {
  await riskyOperation();
} catch (error) {
  logger.error('Operation failed', error);
  // Automatically includes error.name, message, stack
}
```

### Encryption Errors
```typescript
import { EncryptionError, DecryptionError } from '@witylogix/core/encryption';

try {
  const encrypted = crypto.encrypt(data);
} catch (error) {
  if (error instanceof EncryptionError) {
    console.error('Encryption failed:', error.code);
  }
}

try {
  const decrypted = crypto.decrypt(payload);
} catch (error) {
  if (error instanceof DecryptionError) {
    console.error('Decryption failed:', error.code);
    // Could indicate tampered data or wrong key
  }
}
```

## Common Patterns

### Logger with Request Context
```typescript
import { getRequestContext, attachRequestContext } from '@witylogix/core/logging';

app.post('/api/orders', async (request, reply) => {
  const context = getRequestContext(request);
  const log = attachRequestContext(logger, context);

  log.info('Creating order');
  // Includes traceId, userId, tenantId, duration
});
```

### Key Rotation
```typescript
const crypto = createCryptoService();

// Add old keys for decryption
crypto.addKeyVersion('key-2024-q1', oldKeyValue);
crypto.addKeyVersion('key-2024-q2', newerKeyValue);

// New encryptions use the default (latest) key
const payload1 = crypto.encrypt(data); // keyId: 'default'

// But can decrypt with old keys
const oldPayload = { keyId: 'key-2024-q1', ... };
const decrypted = crypto.decrypt(oldPayload); // Works!
```

### Structured Error Logging
```typescript
logger.error('Database query failed', {
  query: 'SELECT * FROM users WHERE id = ?',
  duration: 5000,
  errno: 'ECONNREFUSED',
  retries: 3
});
```
